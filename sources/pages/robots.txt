<script>
(function(){
  var header=document.querySelector('.site-header');
  var nav=document.getElementById('site-nav');
  var toggle=document.querySelector('.nav-toggle');
  if(!header||!nav||!toggle) return;
  function mobile(){ return window.matchMedia('(max-width: 768px)').matches; }
  function closeNav(){ header.classList.remove('nav-open'); nav.hidden=true; toggle.setAttribute('aria-expanded','false'); }
  function openNav(){ header.classList.add('nav-open'); nav.hidden=false; toggle.setAttribute('aria-expanded','true'); }
  function sync(){
    if(mobile()){
      if(!header.classList.contains('nav-open')) closeNav(); else openNav();
    }else{
      header.classList.remove('nav-open'); nav.hidden=false; toggle.setAttribute('aria-expanded','false');
    }
  }
  toggle.type='button';
  toggle.addEventListener('click', function(e){
    e.preventDefault(); e.stopPropagation();
    if(!mobile()) return;
    if(header.classList.contains('nav-open')) closeNav(); else openNav();
  });
  document.addEventListener('click', function(e){ if(mobile() && !header.contains(e.target)) closeNav(); });
  nav.querySelectorAll('a').forEach(function(a){ a.addEventListener('click', function(){ if(mobile()) closeNav(); }); });
  window.addEventListener('resize', sync);
  sync();
})();
</script></body>
